cmake_minimum_required(VERSION 3.22)
set(IDF_TARGET esp32)
set(COMPONENTS main)
if(NOT DEFINED SDKCONFIG)
    set(SDKCONFIG "${CMAKE_CURRENT_BINARY_DIR}/sdkconfig")
endif()
option(CLOCKWISE_WITH_WIFI "Enable Wi-Fi setup and NTP" OFF)
if(CLOCKWISE_WITH_WIFI AND NOT DEFINED SDKCONFIG_DEFAULTS)
    set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;sdkconfig.wifi")
endif()
set(CLOCKWISE_LVGL_PATH "" CACHE PATH "Optional local LVGL 9.6.0 source directory named lvgl")
if(CLOCKWISE_LVGL_PATH)
    set(ENV{IDF_COMPONENT_MANAGER} 0)
    list(APPEND EXTRA_COMPONENT_DIRS "${CLOCKWISE_LVGL_PATH}")
endif()
set(ENV{CW_NATIVE_WIFI} "${CLOCKWISE_WITH_WIFI}")
set(ENV{CW_NATIVE_LVGL} "${CLOCKWISE_LVGL_PATH}")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(clockwise_app_idf)
if(NOT IDF_VERSION_MAJOR EQUAL 6 OR NOT IDF_VERSION_MINOR EQUAL 1)
    message(FATAL_ERROR "Clockwise requires ESP-IDF 6.1")
endif()
